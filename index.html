/** * ======================================================== * TML FINANCE HUB — Backend (Google Apps Script) * Database: Google Sheets (sheet "DAILY REPORT" & "MASTER_CUSTOMER") * ======================================================== * CARA PASANG: * 1. Buka https://sheets.google.com -> buat spreadsheet baru, beri nama "TML Finance Hub DB" * 2. Extensions > Apps Script -> hapus isi default, paste seluruh file ini * 3. Klik Deploy > New deployment > pilih "Web app" * - Execute as: Me * - Who has access: Anyone * 4. Copy "Web app URL" -> tempel ke variabel API_URL di index.html * 5. Jalankan fungsi setup() sekali dari editor (klik Run) untuk membuat sheet & header otomatis * ======================================================== */ const SHEET_NAME = 'DAILY REPORT'; const MASTER_SHEET = 'MASTER_CUSTOMER'; // Urutan kolom di sheet DAILY REPORT (baris 1 = header) const COLS = [ 'ID', 'NO', 'BOOKING_DATE', 'LOADING_DATE', 'NO_BL_DO', 'AJU', 'WO', 'CUSTOMER', 'TUJUAN', 'AREA', 'ZONA', 'TRANSPORTER', 'NOPOL', 'DRIVER_HP', 'SHIPMENT', 'SIZE', 'NO_CONTAINER', 'DEPO', 'PORT', 'REMARK', 'TARIF_VENDOR', 'TARIF_CUSTOMER', 'UJP', 'ADDITIONAL_COST', 'SURAT_JALAN', 'GROSS_MARGIN', // -- Finance tracking -- 'TOP_HARI', 'TGL_INVOICE_TERBIT', 'NO_INVOICE', 'TGL_KIRIM_INVOICE', 'MODE_KIRIM_INVOICE', 'TGL_DITERIMA_CUSTOMER', 'NILAI_REIMBURST', 'DPP_JASA', 'JUMLAH', 'PPN', 'TOTAL_TAGIHAN', 'TGL_DUEDATE', 'TGL_BAYAR', 'KETERANGAN', // -- Kolom terhitung (dihitung ulang tiap request, tidak disimpan statis) -- 'CREATED_AT' ]; /** Jalankan sekali manual dari editor Apps Script untuk inisialisasi sheet */ function setup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) sh = ss.insertSheet(SHEET_NAME); if (sh.getLastRow() === 0) sh.appendRow(COLS); let mc = ss.getSheetByName(MASTER_SHEET); if (!mc) mc = ss.insertSheet(MASTER_SHEET); if (mc.getLastRow() === 0) { mc.appendRow(['CUSTOMER', 'DEFAULT_TOP_HARI']); const defaults = [ ['PT. ANUGERAH TANGKAS TRANSPORTINDO (ATT)', 30], ['PT. DRASENA AGRO NUSA', 30], ['PT. IKS', 30], ['PT. KAIXING INTERNATIONAL LOGISTICS', 30], ['PT. NX LEMO INDONESIA', 30], ['PT. SAMUDERA CARAKA JASA', 30], ['PT. SEKAWAN SINERGI ABADI (SSA)', 30], ['PT. SWADAYA TOTAL PRIMA', 30], ['PT. TRANSPORINDO', 30], ['PT. WINSON', 30], ]; defaults.forEach(r => mc.appendRow(r)); } Logger.log('Setup selesai.'); } function getSheet_() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); } function getMasterSheet_() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(MASTER_SHEET); } function num_(v) { if (v === '' || v === null || v === undefined) return 0; const n = Number(v); return isNaN(n) ? 0 : n; } /** Ambil tanggal hari ini di zona waktu Jakarta (dinormalisasi ke tengah malam UTC agar perbandingan hari akurat) */ function todayJakarta_() { const s = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd'); const [y, m, d] = s.split('-').map(Number); return new Date(Date.UTC(y, m - 1, d)); } /** Normalisasi tanggal apapun (Date object dari Sheet, atau string "yyyy-mm-dd" dari form) ke tanggal kalender Jakarta */ function toUTCDate_(v) { if (!v) return null; const d = new Date(v); if (isNaN(d)) return null; const s = Utilities.formatDate(d, 'Asia/Jakarta', 'yyyy-MM-dd'); const [y, m, day] = s.split('-').map(Number); return new Date(Date.UTC(y, m - 1, day)); } function addDays_(date, days) { const d = new Date(date); d.setUTCDate(d.getUTCDate() + Number(days)); return d; } /** Hitung semua kolom turunan (gross margin, ppn, status barometer, aging, dst) */ function computeDerived_(rec) { const out = {}; out.GROSS_MARGIN = num_(rec.TARIF_CUSTOMER) - num_(rec.UJP); const hasDpp = rec.DPP_JASA !== '' && rec.DPP_JASA !== undefined && rec.DPP_JASA !== null; const jumlah = hasDpp ? num_(rec.DPP_JASA) + num_(rec.NILAI_REIMBURST) : ''; const ppn = hasDpp ? Math.round(num_(rec.DPP_JASA) * 0.011) : ''; const total = hasDpp ? jumlah + ppn : ''; out.JUMLAH = jumlah; out.PPN = ppn; out.TOTAL_TAGIHAN = total; const today = todayJakarta_(); let duedate = null; if (rec.TGL_INVOICE_TERBIT) { const invDate = toUTCDate_(rec.TGL_INVOICE_TERBIT); const top = rec.TOP_HARI; if (top !== '' && top !== undefined && top !== null && !isNaN(Number(top))) { duedate = addDays_(invDate, Number(top)); } else { duedate = invDate; // mis. "CASH" -> jatuh tempo = tanggal invoice } } out.TGL_DUEDATE = duedate ? duedate : ''; // Tanggal bayar hanya dianggap valid (LUNAS) kalau sudah lewat / hari ini -- tanggal di masa depan diabaikan let paidDate = null; if (rec.TGL_BAYAR) { const pd = toUTCDate_(rec.TGL_BAYAR); if (pd && pd <= today) paidDate = pd; } const remark = (rec.REMARK || '').toString().toUpperCase(); let status = ''; let aging = ''; if (!remark) { status = ''; } else if (remark.indexOf('CANCEL') >= 0) { status = '⬜ N/A'; } else if (paidDate) { status = '🟢 LUNAS'; aging = '-'; } else if (!rec.NO_INVOICE) { status = '⚫ Belum Invoice'; // Aging = umur job sejak Booking Date, supaya kelihatan sudah berapa hari belum di-invoice if (rec.BOOKING_DATE) { const bd = toUTCDate_(rec.BOOKING_DATE); if (bd) aging = Math.max(0, Math.round((today - bd) / 86400000)); } } else if (!duedate) { status = '🔵 TerInvoice'; } else { const diffDays = Math.round((duedate - today) / 86400000); if (diffDays >= 0) { status = diffDays <= 7 ? '🟡 Mendekati JT' : '🔵 TerInvoice'; aging = 0; } else { const over = -diffDays; status = over <= 14 ? '🟠 Overdue Ringan' : '🔴 Overdue Berat'; aging = over; } } out.STATUS_BAROMETER = status; out.AGING = aging; // Nilai efektif untuk dashboard: pakai Total Tagihan asli kalau sudah diinvoice (DPP terisi), // kalau belum, pakai estimasi dari Tarif Customer supaya nilainya tetap kelihatan di dashboard. out.EFFECTIVE_VALUE = hasDpp ? total : num_(rec.TARIF_CUSTOMER); return out; } function rowToRecord_(headers, row) { const rec = {}; headers.forEach((h, i) => { rec[h] = row[i] === undefined ? '' : row[i]; }); return rec; } /** Ambil semua data (dipakai untuk tabel Finance Tracking & Input) */ function getData() { const sh = getSheet_(); const lastRow = sh.getLastRow(); if (lastRow < 2) return { ok: true, data: [] }; const headers = sh.getRange(1, 1, 1, COLS.length).getValues()[0]; const values = sh.getRange(2, 1, lastRow - 1, COLS.length).getValues(); const data = values .filter(r => r[0] !== '') // skip baris kosong .map(r => { const rec = rowToRecord_(headers, r); Object.assign(rec, computeDerived_(rec)); return rec; }); return { ok: true, data: data }; } function getCustomers() { const mc = getMasterSheet_(); const lastRow = mc.getLastRow(); if (lastRow < 2) return { ok: true, customers: [] }; const values = mc.getRange(2, 1, lastRow - 1, 2).getValues(); return { ok: true, customers: values.map(r => ({ name: r[0], defaultTop: r[1] })) }; } /** Ringkasan dashboard: KPI + rekap per customer */ function getDashboard() { const res = getData(); const rows = res.data; const kpi = { totalTagihan: 0, outstanding: 0, overdueBerat: 0, belumInvoice: 0, lunas: 0 }; const perCustomer = {}; rows.forEach(r => { const status = r.STATUS_BAROMETER; const total = num_(r.EFFECTIVE_VALUE); if (status && status !== '⬜ N/A') kpi.totalTagihan += total; if (status && status !== '⬜ N/A' && status !== '🟢 LUNAS') kpi.outstanding += total; if (status === '🔴 Overdue Berat') kpi.overdueBerat += total; if (status === '⚫ Belum Invoice') kpi.belumInvoice += 1; if (status === '🟢 LUNAS') kpi.lunas += total; const cust = r.CUSTOMER || '(Tanpa Nama)'; if (!perCustomer[cust]) { perCustomer[cust] = { customer: cust, jumlahShipment: 0, totalTagihan: 0, outstanding: 0, overdueBerat: 0, maxAging: 0 }; } const pc = perCustomer[cust]; pc.jumlahShipment += 1; if (status && status !== '⬜ N/A') pc.totalTagihan += total; if (status && status !== '⬜ N/A' && status !== '🟢 LUNAS') pc.outstanding += total; if (status === '🔴 Overdue Berat') pc.overdueBerat += total; if (typeof r.AGING === 'number') pc.maxAging = Math.max(pc.maxAging, r.AGING); }); const statusCount = {}; ['⚫ Belum Invoice', '🔵 TerInvoice', '🟡 Mendekati JT', '🟠 Overdue Ringan', '🔴 Overdue Berat', '🟢 LUNAS'].forEach(s => { statusCount[s] = rows.filter(r => r.STATUS_BAROMETER === s).length; }); return { ok: true, kpi: kpi, perCustomer: Object.values(perCustomer), statusCount: statusCount }; } /** Tambah entry baru dari Admin Operasional (Input Harian) */ function addEntry(data) { const sh = getSheet_(); const id = Utilities.getUuid(); const lastRow = sh.getLastRow(); const no = lastRow; // baris ke-N sebagai nomor urut sederhana const row = COLS.map(c => { if (c === 'ID') return id; if (c === 'NO') return no; if (c === 'CREATED_AT') return new Date(); if (c === 'GROSS_MARGIN') return num_(data.TARIF_CUSTOMER) - num_(data.UJP); return data[c] !== undefined ? data[c] : ''; }); sh.appendRow(row); return { ok: true, id: id }; } /** Update kolom finance (dipakai tim invoice/finance) berdasarkan ID */ function updateFinance(id, data) { const sh = getSheet_(); const lastRow = sh.getLastRow(); if (lastRow < 2) return { ok: false, error: 'Data kosong' }; const ids = sh.getRange(2, 1, lastRow - 1, 1).getValues().flat(); const idx = ids.indexOf(id); if (idx === -1) return { ok: false, error: 'ID tidak ditemukan' }; const rowNum = idx + 2; const editableFinanceCols = [ 'TOP_HARI', 'TGL_INVOICE_TERBIT', 'NO_INVOICE', 'TGL_KIRIM_INVOICE', 'MODE_KIRIM_INVOICE', 'TGL_DITERIMA_CUSTOMER', 'NILAI_REIMBURST', 'DPP_JASA', 'TGL_BAYAR', 'KETERANGAN', 'REMARK' ]; editableFinanceCols.forEach(c => { if (data[c] !== undefined) { const colIdx = COLS.indexOf(c) + 1; sh.getRange(rowNum, colIdx).setValue(data[c]); } }); return { ok: true }; } /** Hapus 1 baris berdasarkan ID (dipakai tombol Hapus di Finance Tracking) */ function deleteEntry(id) { const sh = getSheet_(); const lastRow = sh.getLastRow(); if (lastRow < 2) return { ok: false, error: 'Data kosong' }; const ids = sh.getRange(2, 1, lastRow - 1, 1).getValues().flat(); const idx = ids.indexOf(id); if (idx === -1) return { ok: false, error: 'ID tidak ditemukan' }; sh.deleteRow(idx + 2); return { ok: true }; } function jsonpOut_(obj, callback) { const json = JSON.stringify(obj); if (callback) { return ContentService.createTextOutput(callback + '(' + json + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { const p = e.parameter; const action = p.action || 'getData'; let result; try { if (action === 'getData') result = getData(); else if (action === 'getDashboard') result = getDashboard(); else if (action === 'getCustomers') result = getCustomers(); else if (action === 'addEntry') result = addEntry(JSON.parse(p.payload)); else if (action === 'updateFinance') result = updateFinance(p.id, JSON.parse(p.payload)); else if (action === 'deleteEntry') result = deleteEntry(p.id); else result = { ok: false, error: 'Unknown action: ' + action }; } catch (err) { result = { ok: false, error: String(err) }; } return jsonpOut_(result, p.callback); }
